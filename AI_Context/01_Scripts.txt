using UnityEngine;
using UnityEngine.SceneManagement;

public class TestBattleManager : MonoBehaviour
{
    [Header("Clear Condition")]
    [SerializeField] private int requiredSkillUses = 3;

    [Header("Reward")]
    [SerializeField] private int clearReward = 100;

    [Header("Scene")]
    [SerializeField] private string townSceneName = "Town";

    private int currentSkillUses = 0;
    private bool isCleared = false;


    public void RegisterTestSkillUse()
    {
        if (isCleared)
            return;

        currentSkillUses++;

        Debug.Log(
            $"[Battle] TestSkill 사용 {currentSkillUses} / {requiredSkillUses}"
        );

        if (currentSkillUses >= requiredSkillUses)
        {
            ClearBattle();
        }
    }


    private void ClearBattle()
    {
        if (isCleared)
            return;

        isCleared = true;

        if (GameSession.Instance == null)
        {
            Debug.LogError("[Battle] GameSession이 없습니다.");
            return;
        }

        GameSession.Instance.Data.testCurrency += clearReward;
        GameSession.Instance.Data.stage1Cleared = true;

        Debug.Log(
            $"[Battle] 클리어 / 보상 +{clearReward}"
        );

        GameSession.Instance.SaveGame();

        Debug.Log("[Battle] 자동 저장 완료");

        SceneManager.LoadScene(townSceneName);
    }
}

using UnityEngine;

public class GoblinAnimationController : MonoBehaviour
{

    [SerializeField] private Rigidbody2D rb;

    [Header("Settings")]

        Animator.StringToHash("IsMoving");

    private static readonly int AttackHash =
        Animator.StringToHash("Attack");

    private static readonly int HitHash =
        Animator.StringToHash("Hit");

    {
        UpdateMovementAnimation();
    }

    private void UpdateMovementAnimation()
    {

            IsMovingHash,
            isMoving
        );
    }

    {
        animator.SetTrigger(
            AttackHash
        );
    }

    public void PlayHit()
    {
        animator.SetTrigger(
            HitHash
        );
    }
}

using System;
using UnityEngine;

[RequireComponent(typeof(Rigidbody2D))]
public class GoblinEnemy : MonoBehaviour
{
    [Header("References")]
    [SerializeField] private SpriteRenderer spriteRenderer;

    [Header("Detection")]
    [SerializeField] private float detectionRange = 7f;

    [Header("Movement")]
    [SerializeField] private float moveSpeed = 2f;
    [SerializeField] private float stopDistance = 0.8f;

    private Rigidbody2D rb;
    private Transform target;

    private Action<GoblinEnemy> releaseAction;

    private bool targetDetected;

    private void Awake()
    {
        rb = GetComponent<Rigidbody2D>();
    }

    private void Update()
    {
        CheckTarget();
        UpdateDirection();
    }

    private void FixedUpdate()
    {
        Move();
    }

    public void Initialize(
        Transform newTarget,
        Action<GoblinEnemy> newReleaseAction)
    {
        target = newTarget;
        releaseAction = newReleaseAction;

        targetDetected = false;

        rb.linearVelocity = Vector2.zero;
    }

    private void CheckTarget()
    {
        if (target == null)
        {
            targetDetected = false;
            return;
        }

        float distance = Vector2.Distance(
            transform.position,
            target.position
        );

        targetDetected = distance <= detectionRange;
    }

    private void Move()
    {
        if (!targetDetected || target == null)
        {
            StopHorizontalMovement();
            return;
        }

        float horizontalDistance =
            target.position.x - transform.position.x;

        if (Mathf.Abs(horizontalDistance) <= stopDistance)
        {
            StopHorizontalMovement();
            return;
        }

        float direction = Mathf.Sign(horizontalDistance);

        rb.linearVelocity = new Vector2(
            direction * moveSpeed,
            rb.linearVelocity.y
        );
    }

    private void StopHorizontalMovement()
    {
        rb.linearVelocity = new Vector2(
            0f,
            rb.linearVelocity.y
        );
    }

    private void UpdateDirection()
    {
        if (!targetDetected || target == null)
            return;

        float horizontalDistance =
            target.position.x - transform.position.x;

        if (horizontalDistance > 0.01f)
        {
            FaceRight();
        }
        else if (horizontalDistance < -0.01f)
        {
            FaceLeft();
        }
    }

    private void FaceRight()
    {
        if (spriteRenderer == null)
            return;

        spriteRenderer.flipX = false;
    }

    private void FaceLeft()
    {
        if (spriteRenderer == null)
            return;

        spriteRenderer.flipX = true;
    }

    public void Despawn()
    {
        releaseAction?.Invoke(this);
    }

    private void OnDrawGizmosSelected()
    {
        Gizmos.DrawWireSphere(
            transform.position,
            detectionRange
        );
    }
}

using System.Collections.Generic;
using TMPro;
using UnityEngine;

public class OptionDisplayController : MonoBehaviour
{
    [Header("Dropdown")]
    [SerializeField] private TMP_Dropdown displayModeDropdown;
    [SerializeField] private TMP_Dropdown resolutionDropdown;

    private readonly List<Vector2Int> resolutions = new();

    private void Start()
    {
        InitializeDisplayModeDropdown();
        InitializeResolutionDropdown();
    }

    private void InitializeDisplayModeDropdown()
    {
        if (displayModeDropdown == null)
            return;

        displayModeDropdown.ClearOptions();
        displayModeDropdown.AddOptions(new List<string>
        {
            "전체화면",
            "창모드"
        });

        displayModeDropdown.SetValueWithoutNotify(Screen.fullScreen ? 0 : 1);
        displayModeDropdown.RefreshShownValue();

        displayModeDropdown.onValueChanged.RemoveAllListeners();
        displayModeDropdown.onValueChanged.AddListener(OnDisplayModeChanged);
    }

    private void InitializeResolutionDropdown()
    {
        if (resolutionDropdown == null)
            return;

        resolutionDropdown.ClearOptions();
        resolutions.Clear();

        List<string> options = new List<string>();
        HashSet<string> added = new HashSet<string>();

        Resolution[] screenResolutions = Screen.resolutions;

        foreach (Resolution r in screenResolutions)
        {
            string key = $"{r.width}x{r.height}";

            if (added.Contains(key))
                continue;

            added.Add(key);
            resolutions.Add(new Vector2Int(r.width, r.height));
            options.Add($"{r.width} x {r.height}");
        }

        if (options.Count == 0)
        {
            resolutions.Add(new Vector2Int(Screen.width, Screen.height));
            options.Add($"{Screen.width} x {Screen.height}");
        }

        int currentIndex = 0;

        for (int i = 0; i < resolutions.Count; i++)
        {
            if (resolutions[i].x == Screen.width &&
                resolutions[i].y == Screen.height)
            {
                currentIndex = i;
                break;
            }
        }

        resolutionDropdown.AddOptions(options);
        resolutionDropdown.SetValueWithoutNotify(currentIndex);
        resolutionDropdown.RefreshShownValue();

        resolutionDropdown.onValueChanged.RemoveAllListeners();
        resolutionDropdown.onValueChanged.AddListener(OnResolutionChanged);
    }

    public void OnDisplayModeChanged(int index)
    {
        bool isFullscreen = index == 0;
        Screen.fullScreen = isFullscreen;

        ApplyCurrentResolution();
    }

    public void OnResolutionChanged(int index)
    {
        ApplyResolution(index);
    }

    private void ApplyCurrentResolution()
    {
        if (resolutionDropdown == null)
            return;

        ApplyResolution(resolutionDropdown.value);
    }

    private void ApplyResolution(int index)
    {
        if (index < 0 || index >= resolutions.Count)
            return;

        Vector2Int resolution = resolutions[index];
        Screen.SetResolution(resolution.x, resolution.y, Screen.fullScreen);
    }
}

using UnityEngine;
using UnityEditor;
using UnityEditor.Animations;

public static class AnimatorTransitionTool
{
    [MenuItem("Tools/Slayers/Animator/Reset All Transitions")]
    private static void ResetAllTransitions()
    {
        AnimatorController controller =
            Selection.activeObject as AnimatorController;

        if (controller == null)
        {
            EditorUtility.DisplayDialog(
                "Animator Transition Tool",
                "Project 창에서 Animator Controller를 선택한 뒤 실행하세요.",
                "OK"
            );

            return;
        }

        int changedCount = 0;

        Undo.RecordObject(
            controller,
            "Reset Animator Transitions"
        );

        foreach (AnimatorControllerLayer layer in controller.layers)
        {
            changedCount += ResetTransitionsRecursive(
                layer.stateMachine
            );
        }

        EditorUtility.SetDirty(controller);

        AssetDatabase.SaveAssets();

        EditorUtility.DisplayDialog(
            "Completed",
            $"{controller.name}\n\n" +
            $"수정된 Transition: {changedCount}개\n\n" +
            "Has Exit Time = OFF\n" +
            "Transition Duration = 0",
            "OK"
        );

        Debug.Log(
            $"[{controller.name}] " +
            $"Reset {changedCount} transitions. " +
            $"Has Exit Time = OFF, Duration = 0."
        );
    }

    private static int ResetTransitionsRecursive(
        AnimatorStateMachine stateMachine)
    {
        int changedCount = 0;

        // 일반 State → State Transition
        foreach (ChildAnimatorState childState
                 in stateMachine.states)
        {
            foreach (AnimatorStateTransition transition
                     in childState.state.transitions)
            {
                bool changed = false;

                if (transition.hasExitTime)
                {
                    transition.hasExitTime = false;
                    changed = true;
                }

                if (transition.duration != 0f)
                {
                    transition.duration = 0f;
                    changed = true;
                }

                if (changed)
                {
                    EditorUtility.SetDirty(transition);
                    changedCount++;
                }
            }
        }

        // Any State → State Transition
        foreach (AnimatorStateTransition transition
                 in stateMachine.anyStateTransitions)
        {
            bool changed = false;

            if (transition.hasExitTime)
            {
                transition.hasExitTime = false;
                changed = true;
            }

            if (transition.duration != 0f)
            {
                transition.duration = 0f;
                changed = true;
            }

            if (changed)
            {
                EditorUtility.SetDirty(transition);
                changedCount++;
            }
        }

        // 서브 State Machine도 재귀적으로 처리
        foreach (ChildAnimatorStateMachine childStateMachine
                 in stateMachine.stateMachines)
        {
            changedCount += ResetTransitionsRecursive(
                childStateMachine.stateMachine
            );
        }

        return changedCount;
    }
}

        string assetsFullPath = Path.Combine(projectRoot, AssetsRoot);
        string outputDirectoryFullPath = Path.Combine(projectRoot, OutputDirectory);

        if (!Directory.Exists(assetsFullPath))
        {
            EditorUtility.DisplayDialog(
                "Export Failed",

        {

                snapshotId,

                projectRoot,
                outputDirectoryFullPath,

                snapshotId,

            );

                projectRoot,
                outputDirectoryFullPath,
                allFiles,
                snapshotId,
                generatedAt
            );

                projectRoot,
                outputDirectoryFullPath,
                allFiles,
                snapshotId,
                generatedAt
            );

                "Snapshot Export Complete",

                $"{OutputDirectory}/{IndexFileName}\n" +
                $"{OutputDirectory}/{ScriptsFileName}\n" +
                $"{OutputDirectory}/{ScenesPrefabsFileName}\n" +
                $"{OutputDirectory}/{GameAssetsFileName}\n" +
                $"{OutputDirectory}/{ExternalAssetsFileName}",
                "OK"
            );

            Debug.Log(
                $"[Slayers Snapshot] Export complete. Snapshot ID: {snapshotId}"
            );
        }
        catch (Exception ex)
        {
            Debug.LogException(ex);

            EditorUtility.DisplayDialog(
                "Export Failed",
                ex.Message,
                "OK"

    private static void WriteExtensionSummary(
        StringBuilder builder,
        string[] allFiles)
    {
        builder.AppendLine("==================================================");
        builder.AppendLine("EXTENSION SUMMARY");
        builder.AppendLine("==================================================");
        builder.AppendLine();

        var groups = allFiles
            .GroupBy(path =>
            {
                string extension = Path.GetExtension(path);

                return string.IsNullOrEmpty(extension)
                    ? "(no extension)"
                    : extension.ToLowerInvariant();
            })
            .OrderBy(group => group.Key);

        foreach (var group in groups)
        {
            builder.AppendLine($"{group.Key} : {group.Count()}");
        }
    }

    private static void AppendTextFile(
        StringBuilder builder,
        string projectRoot,
        string fullPath)
    {
        string relativePath = GetRelativePath(projectRoot, fullPath);

        builder.AppendLine();
        builder.AppendLine(
            "--------------------------------------------------"
        );
        builder.AppendLine($"FILE: {relativePath}");
        builder.AppendLine(
            "--------------------------------------------------"
        );

        try
        {
            FileInfo info = new FileInfo(fullPath);

            builder.AppendLine($"Size: {FormatFileSize(info.Length)}");
            builder.AppendLine(
                $"Last Modified: {info.LastWriteTime:yyyy-MM-dd HH:mm:ss}"
            );
            builder.AppendLine();

            string content = File.ReadAllText(
                fullPath,
                Encoding.UTF8
            );

            builder.AppendLine(content);
        }
        catch (Exception ex)
        {
            builder.AppendLine(
                $"[READ ERROR] {ex.GetType().Name}: {ex.Message}"
            );
        }

        builder.AppendLine();
    }

    private static bool IsExternalAsset(string relativePath)
    {
        return relativePath.StartsWith(
            ExternalAssetsRoot,
            StringComparison.OrdinalIgnoreCase
        );
    }

    private static string GetRelativePath(
        string projectRoot,
        string fullPath)
    {
        string relativePath = Path.GetRelativePath(
            projectRoot,
            fullPath
        );

        return relativePath.Replace('\\', '/');
    }

    private static string FormatFileSize(long bytes)
    {
        if (bytes >= 1024L * 1024L)
        {
            return $"{bytes / (1024d * 1024d):0.00} MB";
        }

        if (bytes >= 1024L)
        {
            return $"{bytes / 1024d:0.00} KB";
        }

        return $"{bytes} B";
    }

    private static void WriteFile(
        string path,
        StringBuilder builder)
    {
        File.WriteAllText(
            path,
            builder.ToString(),
            new UTF8Encoding(false)
        );
    }

using UnityEngine;
using UnityEditor;
using UnityEditor.U2D.Sprites;

public class SpritePivotBatchTool : EditorWindow
{
    private Texture2D targetTexture;

    private PivotPreset pivotPreset = PivotPreset.BottomCenter;
    private Vector2 customPivot = new Vector2(0.5f, 0f);

    private enum PivotPreset
    {
        Center,
        BottomCenter,
        Custom
    }

    [MenuItem("Tools/Slayers/Sprite Pivot Batch Tool")]
    private static void OpenWindow()
    {
        GetWindow<SpritePivotBatchTool>("Sprite Pivot Batch Tool");
    }

    private void OnSelectionChange()
    {
        if (Selection.activeObject is Texture2D texture)
        {
            targetTexture = texture;
            Repaint();
        }
    }

    private void OnGUI()
    {
        EditorGUILayout.Space();
        EditorGUILayout.LabelField("Batch Change Sprite Pivots", EditorStyles.boldLabel);
        EditorGUILayout.Space();

        targetTexture = (Texture2D)EditorGUILayout.ObjectField(
            "Target Texture",
            targetTexture,
            typeof(Texture2D),
            false
        );

        pivotPreset = (PivotPreset)EditorGUILayout.EnumPopup("Pivot Preset", pivotPreset);

        if (pivotPreset == PivotPreset.Custom)
        {
            customPivot = EditorGUILayout.Vector2Field("Custom Pivot", customPivot);
            customPivot.x = Mathf.Clamp01(customPivot.x);
            customPivot.y = Mathf.Clamp01(customPivot.y);
        }

        EditorGUILayout.Space();

        EditorGUILayout.HelpBox(
            "This tool changes the pivot of all sprites inside the selected sprite sheet.\n" +
            "Recommended for characters: BottomCenter",
            MessageType.Info
        );

        EditorGUILayout.Space();

        GUI.enabled = targetTexture != null;

        if (GUILayout.Button("Apply Pivot To All Sprites", GUILayout.Height(30)))
        {
            ApplyPivotToAllSprites();
        }

        GUI.enabled = true;
    }

    private void ApplyPivotToAllSprites()
    {
        if (targetTexture == null)
        {
            EditorUtility.DisplayDialog("Error", "Please assign a target texture.", "OK");
            return;
        }

        string assetPath = AssetDatabase.GetAssetPath(targetTexture);
        TextureImporter importer = AssetImporter.GetAtPath(assetPath) as TextureImporter;

        if (importer == null)
        {
            EditorUtility.DisplayDialog("Error", "Could not find TextureImporter.", "OK");
            return;
        }

        if (importer.textureType != TextureImporterType.Sprite)
        {
            EditorUtility.DisplayDialog("Error", "Selected texture is not imported as Sprite.", "OK");
            return;
        }

        if (importer.spriteImportMode != SpriteImportMode.Multiple)
        {
            bool continueAnyway = EditorUtility.DisplayDialog(
                "Warning",
                "The selected texture is not set to Multiple sprite mode.\n\nContinue anyway?",
                "Continue",
                "Cancel"
            );

            if (!continueAnyway)
                return;
        }

        SpriteDataProviderFactories factory = new SpriteDataProviderFactories();
        factory.Init();

        ISpriteEditorDataProvider dataProvider = factory.GetSpriteEditorDataProviderFromObject(importer);

        if (dataProvider == null)
        {
            EditorUtility.DisplayDialog("Error", "Could not get sprite data provider.", "OK");
            return;
        }

        dataProvider.InitSpriteEditorDataProvider();

        SpriteRect[] spriteRects = dataProvider.GetSpriteRects();

        if (spriteRects == null || spriteRects.Length == 0)
        {
            EditorUtility.DisplayDialog("Error", "No sprite rects found in this texture.", "OK");
            return;
        }

        for (int i = 0; i < spriteRects.Length; i++)
        {
            switch (pivotPreset)
            {
                case PivotPreset.Center:
                    spriteRects[i].alignment = SpriteAlignment.Center;
                    spriteRects[i].pivot = new Vector2(0.5f, 0.5f);
                    break;

                case PivotPreset.BottomCenter:
                    spriteRects[i].alignment = SpriteAlignment.BottomCenter;
                    spriteRects[i].pivot = new Vector2(0.5f, 0f);
                    break;

                case PivotPreset.Custom:
                    spriteRects[i].alignment = SpriteAlignment.Custom;
                    spriteRects[i].pivot = customPivot;
                    break;
            }
        }

        dataProvider.SetSpriteRects(spriteRects);
        dataProvider.Apply();

        importer.SaveAndReimport();

        EditorUtility.DisplayDialog(
            "Completed",
            $"Updated pivot for {spriteRects.Length} sprites.",
            "OK"
        );
    }

using UnityEngine;

[RequireComponent(typeof(Rigidbody2D))]
[RequireComponent(typeof(CapsuleCollider2D))]
public class CharacterMotor2D : MonoBehaviour
{
    [Header("References")]
    [SerializeField] private PlayerInputReader inputReader;
    [SerializeField] private LayerMask groundLayer;

    [Header("Movement")]
    [SerializeField] private float moveSpeed = 5f;

    [Header("Jump")]
    [SerializeField] private float jumpForce = 10f;

    [Header("Ground Check")]
    [SerializeField] private float groundCheckDistance = 0.08f;

    private Rigidbody2D rb;
    private CapsuleCollider2D capsuleCollider;

    private bool isGrounded;

    // 나중에 스킬/넉백과 분리하기 위한 속도
    private float inputVelocityX;
    private Vector2 skillVelocity;
    private Vector2 knockbackVelocity;

    public bool IsGrounded => isGrounded;
    public float MoveInputX => inputReader.MoveInput.x;

    private void Awake()
    {
        rb = GetComponent<Rigidbody2D>();
        capsuleCollider = GetComponent<CapsuleCollider2D>();
    }

    private void Update()
    {
        HandleInput();
        HandleJump();
    }

    private void FixedUpdate()
    {
        CheckGround();
        ApplyVelocity();
    }

    private void HandleInput()
    {
        inputVelocityX = inputReader.MoveInput.x * moveSpeed;

        if (inputReader.MoveInput.x > 0.01f)
        {
            FaceRight();
        }
        else if (inputReader.MoveInput.x < -0.01f)
        {
            FaceLeft();
        }
    }

    private void HandleJump()
    {
        if (!inputReader.ConsumeJump())
            return;

        if (!isGrounded)
            return;

        isGrounded = false;

        rb.linearVelocity = new Vector2(
            rb.linearVelocity.x,
            0f
        );

        rb.AddForce(
            Vector2.up * jumpForce,
            ForceMode2D.Impulse
        );
    }

    private void ApplyVelocity()
    {
        float finalX =
            inputVelocityX +
            skillVelocity.x +
            knockbackVelocity.x;

        rb.linearVelocity = new Vector2(
            finalX,
            rb.linearVelocity.y
        );
    }

    private void CheckGround()
    {
        Bounds bounds = capsuleCollider.bounds;

        Vector2 checkSize = new Vector2(
            bounds.size.x * 0.8f,
            0.05f
        );

        Vector2 checkPosition = new Vector2(
            bounds.center.x,
            bounds.min.y + 0.02f
        );

        RaycastHit2D hit = Physics2D.BoxCast(
            checkPosition,
            checkSize,
            0f,
            Vector2.down,
            groundCheckDistance,
            groundLayer
        );

        isGrounded = hit.collider != null;
    }

    private void FaceRight()
    {
        transform.localScale = new Vector3(
            Mathf.Abs(transform.localScale.x),
            transform.localScale.y,
            transform.localScale.z
        );
    }

    private void FaceLeft()
    {
        transform.localScale = new Vector3(
            -Mathf.Abs(transform.localScale.x),
            transform.localScale.y,
            transform.localScale.z
        );
    }

    private void OnDrawGizmosSelected()
    {
        CapsuleCollider2D capsule =
            GetComponent<CapsuleCollider2D>();

        if (capsule == null)
            return;

        Bounds bounds = capsule.bounds;

        Vector2 checkSize = new Vector2(
            bounds.size.x * 0.8f,
            0.05f
        );

        Vector2 checkPosition = new Vector2(
            bounds.center.x,
            bounds.min.y + 0.02f
        );

        Gizmos.DrawWireCube(
            checkPosition +
            Vector2.down * groundCheckDistance,
            checkSize
        );
    }
}

using UnityEngine;

public class PlayerAnimationController : MonoBehaviour
{
    [Header("References")]
    [SerializeField] private Animator animator;
    [SerializeField] private CharacterMotor2D motor;
    [SerializeField] private Rigidbody2D rb;

    [Header("Animation Settings")]
    [SerializeField] private float moveThreshold = 0.01f;
    [SerializeField] private float verticalThreshold = 0.05f;

    // 방향 전환 시 순간적인 0 입력 때문에
    // Stand로 바뀌는 것을 방지하는 시간
    [SerializeField] private float moveGraceTime = 0.08f;

    private float moveGraceTimer;

    private static readonly int SpeedHash =
        Animator.StringToHash("Speed");

    private static readonly int IsGroundedHash =
        Animator.StringToHash("IsGrounded");

    private static readonly int VerticalSpeedHash =
        Animator.StringToHash("VerticalSpeed");

    private void Update()
    {
        UpdateAnimatorParameters();
    }

    private void UpdateAnimatorParameters()
    {
        float moveInput = Mathf.Abs(motor.MoveInputX);
        float verticalSpeed = rb.linearVelocity.y;

        float animationSpeed;

        // 이동 입력이 존재하면
        // 달리기 판정을 일정 시간 유지한다.
        if (moveInput > moveThreshold)
        {
            moveGraceTimer = moveGraceTime;
            animationSpeed = moveInput;
        }
        else
        {
            moveGraceTimer -= Time.deltaTime;

            if (moveGraceTimer > 0f)
            {
                animationSpeed = 1f;
            }
            else
            {
                animationSpeed = 0f;
            }
        }

        if (Mathf.Abs(verticalSpeed) < verticalThreshold)
        {
            verticalSpeed = 0f;
        }

        animator.SetFloat(
            SpeedHash,
            animationSpeed
        );

        animator.SetBool(
            IsGroundedHash,
            motor.IsGrounded
        );

        animator.SetFloat(
            VerticalSpeedHash,
            verticalSpeed
        );
    }
}

using System;
using UnityEngine;

[Serializable]
public class StatGrowth
{
    [SerializeField] private float level1Value;
    [SerializeField] private float growthPerLevel;

    public float Level1Value => level1Value;
    public float GrowthPerLevel => growthPerLevel;

    public StatGrowth(float level1Value, float growthPerLevel)
    {
        this.level1Value = level1Value;
        this.growthPerLevel = NormalizeGrowth(growthPerLevel);
    }

    public float GetValue(int level)
    {
        level = Mathf.Max(1, level);

        return level1Value +
               growthPerLevel * (level - 1);
    }

    public float GetLevel50Value()
    {
        return GetValue(50);
    }

    public void Validate()
    {
        growthPerLevel = NormalizeGrowth(growthPerLevel);
    }

    private static float NormalizeGrowth(float value)
    {
        // 성장하지 않는 스탯
        if (value == 0f)
            return 0f;

        // 양수 성장량의 최소값
        if (value > 0f && value < 0.05f)
            value = 0.05f;

        // 소수 둘째 자리까지
        value = Mathf.Round(value * 100f) / 100f;

        return value;
    }
}

public class InputBindingManager : MonoBehaviour
{
    [SerializeField] private InputActionAsset inputActions;

    private InputAction testSkillAction;


    private void Awake()
    {
        if (inputActions == null)
        {
            Debug.LogError("[InputBinding] InputActionAsset이 없습니다.");
            return;
        }

        testSkillAction = inputActions.FindAction("player/TestSkill");

        if (testSkillAction == null)
        {
            Debug.LogError("[InputBinding] player/TestSkill을 찾을 수 없습니다.");
            return;
        }

        ApplySavedBindings();
    }


    public void BindTestSkillToQ()
    {
        BindTestSkill("<Keyboard>/q");
    }


    public void BindTestSkillToE()
    {
        BindTestSkill("<Keyboard>/e");
    }


    private void BindTestSkill(string controlPath)
    {
        if (testSkillAction == null)
            return;

        // 현재 TestSkill에는 단일 키 Binding 하나가 있으므로 index 0
        testSkillAction.ApplyBindingOverride(0, controlPath);

        SaveBindingsToGameData();

        Debug.Log(
            $"[InputBinding] TestSkill 변경 완료 : {controlPath}"
        );
    }


    private void SaveBindingsToGameData()
    {
        if (GameSession.Instance == null)
        {
            Debug.LogError("[InputBinding] GameSession이 없습니다.");
            return;
        }

        string json = inputActions.SaveBindingOverridesAsJson();

        GameSession.Instance.Data.controlBindingsJson = json;

        Debug.Log("[InputBinding] 키 설정을 GameData에 기록");
    }


    private void ApplySavedBindings()
    {
        if (GameSession.Instance == null)
            return;

        string json =
            GameSession.Instance.Data.controlBindingsJson;

        if (string.IsNullOrEmpty(json))
            return;

        inputActions.LoadBindingOverridesFromJson(json);

        Debug.Log("[InputBinding] 저장된 키 설정을 Town에 적용");
    }
}

public class GoblinPool : MonoBehaviour

    [Header("Spawn")]
    [SerializeField] private Transform spawnPoint;
    [SerializeField] private Transform player;

    [Header("Pool Settings")]
    [SerializeField] private int defaultCapacity = 1;
    [SerializeField] private int maxPoolSize = 1;

    private ObjectPool<GoblinEnemy> pool;

    private GoblinEnemy activeGoblin;

    private void Awake()
    {
        pool = new ObjectPool<GoblinEnemy>(
            CreateGoblin,
            OnTakeFromPool,
            OnReturnedToPool,
            OnDestroyPoolObject,
            true,
            defaultCapacity,
            maxPoolSize
        );
    }

    public void SpawnGoblin()
    {
        // 현재 활성화된 고블린이 있다면
        // 추가 소환하지 않는다.
        if (activeGoblin != null &&
            activeGoblin.gameObject.activeSelf)
        {
            return;
        }

        activeGoblin = pool.Get();

        Vector3 spawnPosition =
            spawnPoint != null
                ? spawnPoint.position
                : transform.position;

        activeGoblin.transform.position =
            spawnPosition;

        activeGoblin.Initialize(
            player,
            ReleaseGoblin
        );
    }

    private GoblinEnemy CreateGoblin()
    {
        GoblinEnemy goblin = Instantiate(
            goblinPrefab
        );

        goblin.gameObject.SetActive(false);

        return goblin;
    }

    private void OnTakeFromPool(
        GoblinEnemy goblin)
    {
        goblin.gameObject.SetActive(true);
    }

    private void OnReturnedToPool(
        GoblinEnemy goblin)
    {
        goblin.gameObject.SetActive(false);
    }

    private void OnDestroyPoolObject(
        GoblinEnemy goblin)
    {
        Destroy(goblin.gameObject);
    }

    private void ReleaseGoblin(
        GoblinEnemy goblin)
    {
        if (activeGoblin == goblin)
        {
            activeGoblin = null;
        }

        pool.Release(goblin);
    }
}

using System.IO;
using UnityEngine;

public static class SaveManager
{
    private const string SaveFileName = "save.json";

    private static string SavePath
    {
        get
        {
            return Path.Combine(
                Application.persistentDataPath,
                SaveFileName
            );
        }
    }

    public static void Save(GameData data)
    {
        if (data == null)
        {
            Debug.LogError("[SaveManager] 저장할 GameData가 없습니다.");
            return;
        }

        string json = JsonUtility.ToJson(data, true);

        File.WriteAllText(SavePath, json);

        Debug.Log($"[SaveManager] 저장 완료\n{SavePath}");
    }

    public static GameData Load()
    {
        if (!File.Exists(SavePath))
        {
            Debug.Log("[SaveManager] 세이브 파일이 없습니다.");
            return null;
        }

        string json = File.ReadAllText(SavePath);

        GameData data = JsonUtility.FromJson<GameData>(json);

        Debug.Log($"[SaveManager] 로드 완료\n{SavePath}");

        return data;
    }

    public static bool HasSaveFile()
    {
        return File.Exists(SavePath);
    }

    public static void DeleteSave()
    {
        if (!File.Exists(SavePath))
        {
            return;
        }

        File.Delete(SavePath);

        Debug.Log("[SaveManager] 세이브 파일 삭제 완료");
    }
}
